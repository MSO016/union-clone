CREATE INDEX idx_clients_tipo_cliente

CREATE TABLE clients (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            name TEXT NOT NULL,
            phone TEXT,
            email TEXT,
            address TEXT,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            user_id INTEGER NOT NULL, tipo_cliente TEXT DEFAULT 'pessoa_fisica', observacoes TEXT DEFAULT '',
            FOREIGN KEY (user_id) REFERENCES users(id)
        );

CREATE TABLE knowledge_base (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            device_type TEXT NOT NULL,
            brand TEXT NOT NULL,
            model TEXT NOT NULL,
            issue_description TEXT NOT NULL,
            solution TEXT NOT NULL,
            steps_taken TEXT NOT NULL,
            success_rate INTEGER,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            user_id INTEGER NOT NULL,
            FOREIGN KEY (user_id) REFERENCES users(id)
        );
CREATE TABLE knowledge_base_attachments (id INTEGER PRIMARY KEY AUTOINCREMENT, knowledge_base_id INTEGER, file_path TEXT, file_type TEXT, description TEXT, FOREIGN KEY (knowledge_base_id) REFERENCES knowledge_base(id));
CREATE TABLE service_order_photos (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            service_order_id INTEGER NOT NULL,
            photo_path TEXT NOT NULL,
            description TEXT,
            upload_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            FOREIGN KEY (service_order_id) REFERENCES service_orders(id)
        );
CREATE TABLE service_orders (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            client_id INTEGER NOT NULL,
            device_type TEXT,
            brand TEXT,
            model TEXT,
            serial_number TEXT,
            reported_issue TEXT,
            initial_diagnosis TEXT,
            status TEXT NOT NULL,
            solution TEXT,
            steps_taken TEXT,
            failure_reason TEXT,
            entry_date TIMESTAMP DEFAULT (datetime('now', 'localtime')),
            completion_date TIMESTAMP,
            technician_notes TEXT,
            service_value DECIMAL(10,2),
            user_id INTEGER NOT NULL, completion_photo TEXT, original_os_id INTEGER, diagnosis TEXT,
            FOREIGN KEY (client_id) REFERENCES clients(id),
            FOREIGN KEY (user_id) REFERENCES users(id)
        );

CREATE INDEX idx_cache_valido ON cache_servidor_arquivos(cache_valido_ate);
CREATE INDEX idx_categoria ON cache_servidor_arquivos(categoria);
CREATE INDEX idx_downloads_desc ON cache_servidor_arquivos(downloads_count DESC);
CREATE INDEX idx_gratuito ON cache_servidor_arquivos(is_gratuito);
CREATE INDEX idx_marca ON cache_servidor_arquivos(marca);
CREATE INDEX idx_marca_categoria ON cache_servidor_arquivos(marca, categoria);
CREATE INDEX idx_titulo_search ON cache_servidor_arquivos(LOWER(titulo));
CREATE TABLE cache_config (
                        chave TEXT PRIMARY KEY,
                        valor TEXT NOT NULL,
                        data_atualizacao DATETIME DEFAULT CURRENT_TIMESTAMP
                    );
CREATE TABLE cache_marcas_resumo (
                        marca TEXT PRIMARY KEY,
                        total_arquivos INTEGER DEFAULT 0,
                        total_gratuitos INTEGER DEFAULT 0,
                        total_premiums INTEGER DEFAULT 0,
                        total_downloads INTEGER DEFAULT 0,
                        categorias_json TEXT,
                        cache_valido_ate DATETIME
                    );
CREATE TABLE cache_servidor_arquivos (
                        id INTEGER PRIMARY KEY,
                        marca TEXT NOT NULL,
                        categoria TEXT NOT NULL,
                        titulo TEXT NOT NULL,
                        descricao TEXT,
                        versao TEXT,
                        tamanho_arquivo INTEGER,
                        tipo_arquivo TEXT,
                        data_upload DATETIME,
                        downloads_count INTEGER DEFAULT 0,
                        is_gratuito BOOLEAN DEFAULT 0,
                        is_premium BOOLEAN DEFAULT 0,
                        tags TEXT,
                        rating REAL DEFAULT 0.0,
                        status TEXT DEFAULT 'ativo',
                        preco DECIMAL(10,2),
                        autor TEXT,
                        
                        -- Controle do cache
                        cache_timestamp DATETIME DEFAULT CURRENT_TIMESTAMP,
                        cache_valido_ate DATETIME,
                        hash_metadados TEXT,
                        
                        UNIQUE(id)
                    );
CREATE TABLE cache_sync_control (
                        id INTEGER PRIMARY KEY AUTOINCREMENT,
                        tipo_sync TEXT NOT NULL,
                        timestamp_inicio DATETIME DEFAULT CURRENT_TIMESTAMP,
                        timestamp_fim DATETIME,
                        status TEXT DEFAULT 'em_andamento',
                        registros_processados INTEGER DEFAULT 0,
                        erros INTEGER DEFAULT 0,
                        duracao_segundos REAL DEFAULT 0.0
                    );
CREATE TABLE sqlite_sequence(name,seq);
